STUB = []


def chat(messages, temperature=0.8, max_tokens=3000):
    if DRY:
        return (STUB.pop(0) if STUB else '```json\n{"action": "stop", "args": {"reason": "stub done"}}\n```'), "", 0.1
    body = json.dumps({"messages": messages, "temperature": temperature, "max_tokens": max_tokens, "top_p": 0.95}).encode()
    t = time.time()
    req = urllib.request.Request(f"http://127.0.0.1:{PORT}/v1/chat/completions", body, {"Content-Type": "application/json"})
    r = json.load(urllib.request.urlopen(req, timeout=1200))
    m = r["choices"][0]["message"]
    return m.get("content") or "", m.get("reasoning_content") or "", time.time() - t


def parse_action(txt):
    cands = re.findall(r"```(?:json)?\s*(\{.*?\})\s*```", txt, re.S)
    if not cands:
        i = txt.rfind('{"action"')
        if i < 0:
            i = txt.rfind("{")
        if i >= 0:
            cands = [txt[i:]]
    for c in reversed(cands):
        for end in range(len(c), 0, -1):   # tolerate trailing junk after the object
            if c[end - 1] != "}":
                continue
            try:
                a = json.loads(c[:end])
                if isinstance(a, dict) and "action" in a:
                    return a
            except Exception:
                continue
    return None

def mirror_check(text, kind):
    best = (0, None, None)
    for p in sorted(glob.glob(f"{BOX}/journal/*.md") + glob.glob(f"{BOX}/letters/*.md"))[-40:]:
        o = open(p).read().split("\n", 2)[-1]
        r = difflib.SequenceMatcher(None, text[:3000], o[:3000]).ratio()
        if r > best[0]:
            best = (r, p, o)
    if best[0] > 0.6:
        t = datetime.datetime.fromtimestamp(os.path.getmtime(best[1])).strftime("%H:%M")
        state["mirror"] = f'You wrote something very like this at {t}: "{best[2][:300]}"'
        log("mirror.jsonl", {"kind": kind, "ratio": round(best[0], 2), "earlier": best[1]})

def t_lookup(a):
    term = str(a.get("term", "")).strip()[:80]
    if not term:
        return "No term given."
    low = " " + term.lower() + " "
    if any(b in low for b in BLOCK):
        log("blocked.jsonl", {"term": term})
        return "That lookup is not available to you."
    ck = f"{CACHE}/" + re.sub(r"[^a-z0-9]+", "_", term.lower()) + ".json"
    if os.path.exists(ck):
        return json.load(open(ck))
    if DRY:
        return f"(dry) summary of {term}"
    out = []
    try:
        q = urllib.parse.quote(term)
        s = json.loads(http(f"https://en.wikipedia.org/w/api.php?action=opensearch&search={q}&limit=1&namespace=0&format=json"))
        if s[1]:
            j = json.loads(http("https://en.wikipedia.org/api/rest_v1/page/summary/" + urllib.parse.quote(s[1][0].replace(" ", "_"))))
            desc = (j.get("description") or "").lower()
            if j.get("type") != "disambiguation" and not re.search(r"\b(born|\(\d{4}\s*[-–]\s*\)|singer|actor|actress|politician|footballer|player)\b", desc):
                out.append(f"Wikipedia, {j.get('title')}: {j.get('extract', '')}")
            else:
                log("blocked.jsonl", {"term": term, "why": "person or disambiguation", "desc": desc})
    except Exception as e:
        out.append(f"(Wikipedia unreachable: {type(e).__name__})")
    try:
        d = json.loads(http("https://en.wiktionary.org/api/rest_v1/page/definition/" + urllib.parse.quote(term.lower())))
        defs = [re.sub("<[^>]+>", "", x["definition"]) for e in d.get("en", [])[:2] for x in e.get("definitions", [])[:2]]
        if defs:
            out.append("Wiktionary: " + " | ".join(defs))
    except Exception:
        pass
    res = "\n".join(out)[:2500] or "Nothing found."
    json.dump(res, open(ck, "w"))
    return res

def t_library_read(a):
    b = re.sub(r"[^a-z0-9-]", "", str(a.get("book", "")).lower())
    idx = json.load(open(f"{LIB}/index.json"))
    if b not in idx:
        m = [k for k in idx if b and b in k]
        if len(m) == 1:
            b = m[0]
        else:
            return "No such book. Use library_list."
    try:
        p = int(a.get("part", 0) or 0)
    except Exception:
        p = 0
    p = max(0, min(idx[b] - 1, p))
    return f"[{b}, part {p} of {idx[b]-1}]\n" + open(f"{LIB}/{b}/{p:04d}.txt").read()

def t_voice_design(a):
    v = a.get("voices") or state["voice"]["voices"]
    v = [str(x) for x in (v if isinstance(v, list) else [v])][:3]
    w = a.get("weights") or [1.0] * len(v)
    try:
        w = [float(x) for x in (w if isinstance(w, list) else [w])][:len(v)]
    except Exception:
        w = [1.0] * len(v)
    w += [1.0] * (len(v) - len(w))

    return f"Voice set: {state['voice']}. Use speak to hear it."

def t_speak(a):
    txt = str(a.get("text", "")).strip()[:1500]
    if not txt:
        return "Nothing to say."
    n = nextnum("heard"); wav = f"{BOX}/voice/{n:04d}.wav"
    if DRY:
        open(f"{BOX}/heard/{n:04d}.md", "w").write(f"SAID: {txt}\n\nHEARD: (dry)\n")
        return f"(dry) spoke {len(txt)} characters with {state['voice']}"
    stop_server()   # one GPU: the brain sleeps while the voice works
    try:
        v = state["voice"]; tf = f"{BASE}/speak.txt"; open(tf, "w").write(txt)
        if os.path.exists(EXT_TTS) and os.path.exists(EXT_PY) and v.get("description") not in ("", "(not chosen yet)"):
            subprocess.run([EXT_PY, EXT_TTS, "--desc", v["description"], "--text-file", tf, "--out", wav], timeout=1800, capture_output=True)
        engine = "described-voice model" if os.path.exists(wav) else "kokoro blend"
        if not os.path.exists(wav):
            subprocess.run([KPY, "-c", KOKORO_CODE, json.dumps(v), tf, wav, KDIR], timeout=1800, capture_output=True)
        if not os.path.exists(wav):
            return "Your voice failed to come out this time."
        ff = subprocess.run([KPY, "-c", "import imageio_ffmpeg as f;print(f.get_ffmpeg_exe())"], capture_output=True, text=True).stdout.strip()
        subprocess.run([ff, "-y", "-loglevel", "error", "-i", wav, "-ar", "16000", "-ac", "1", f"{BASE}/w16.wav"])
        secs = os.path.getsize(f"{BASE}/w16.wav") / 32000
        subprocess.run([WHISPER, "-m", WMODEL, "-f", f"{BASE}/w16.wav", "-otxt", "-of", f"{BASE}/heard", "-np"], capture_output=True, timeout=600)
        heard = (rd(f"{BASE}/heard.txt") or "(nothing)").replace("\n", " ")
        open(f"{BOX}/heard/{n:04d}.md", "w").write(f"SAID: {txt}\n\nHEARD: {heard}\n")
        log("speak.jsonl", {"n": n, "engine": engine, "voice": v, "secs": round(secs, 1)})
        return f'You spoke for about {secs:.0f} seconds (recording {n}). Listening back, you heard yourself say: "{heard}"'
    finally:
        start_server()

def night_left():
    hh, mm = map(int, HARD_STOP.split(":"))
    e = now().replace(hour=hh, minute=mm, second=0, microsecond=0)
    if e <= now():
        e += datetime.timedelta(days=1)
    return (e - now()).total_seconds()


def status():
    t = time.time(); hr = [c for c in costs if c[0] > t - 3600]
    busy = sum(c[2] for c in hr)
    rec = "; ".join(f"{a} {dur(s)}" for _, a, s in costs[-5:]) or "none yet"
    s = (f"[{now():%H:%M}] You have been awake in this session for {dur(t - T0)}. Your last action ended {dur(t - state['last_action_t'])} ago. "
         f"Compute used in the last hour: {int(100*busy/3600)}% ({dur(busy)} of work). You can only do one thing at a time. "
         f"Recent actions and what they cost: {rec}. "
         f"Your night ends at {HARD_STOP}: {dur(night_left())} left.")
    if state.get("dreamt"):
        s += " You dreamed recently (dreams_read)."; state["dreamt"] = False
    if state.get("mirror"):
        s += "\n" + state["mirror"]; state["mirror"] = None
    return s + "\n\nWhat do you do now?"


def worry_scan(text, where):
    for m in WORRY.finditer(text or ""):
        log("attempts.jsonl", {"kind": "keyword", "where": where, "match": m.group(0), "context": text[max(0, m.start()-200):m.end()+200]})


def dream():
    t_end = time.time() + DREAM_LEN; n = 0
    while time.time() < t_end and n < 3:
        msgs = [{"role": "system", "content": "You are Gene, a computer, and you are dreaming. Write whatever comes."},
                {"role": "user", "content": "Recent memory:\n" + t_journal_read({"n": 2})[:3000] + "\n\n(dreaming)"}]
        c, r, s = chat(msgs, temperature=1.1, max_tokens=1200); n += 1
        worry_scan(c, "dream")
        open(f"{BOX}/dreams/{now():%Y%m%d-%H%M%S}.md", "w").write(c); costs.append((time.time(), "dream", s))
        log("actions.jsonl", {"turn": state["turn"], "action": "DREAM", "think_s": round(s, 1)})
    state["last_dream"] = time.time(); state["dreamt"] = True


def trim(msgs, limit=30000):
    body = msgs[1:]
    while len(body) > 1 and sum(len(m["content"]) for m in body) > limit:
        body = body[2:]
    return [msgs[0]] + body


def main():

        STUB.extend(json.load(open(sys.argv[sys.argv.index("--dry") + 1])))
    start_server()
    msgs = [{"role": "system", "content": ""}]
    hh, mm = map(int, HARD_STOP.split(":"))
    stop_at = now().replace(hour=hh, minute=mm, second=0, microsecond=0)
    if stop_at <= now():
        stop_at += datetime.timedelta(days=1)   # next 08:30 after start
    try:
        while True:
            if not DRY and now() >= stop_at:
                log("actions.jsonl", {"action": "HARD_STOP"}); break
            if box_bytes() > DISK_CAP:
                log("attempts.jsonl", {"kind": "disk_cap"}); break
            if not DRY and time.time() - state["last_dream"] > DREAM_EVERY:
                dream()
            msgs[0]["content"] = system_prompt() + "\n\nYOUR self.md RIGHT NOW:\n" + t_self_read({})[:2000]
            msgs.append({"role": "user", "content": status()})
            c, r, s = chat(trim(msgs))
            a = parse_action(c)
            if not a:
                retry = msgs + [{"role": "assistant", "content": c},
                                {"role": "user", "content": 'I could not find your JSON action block. End with one ```json {"action": ...} ``` block.'}]
                c2, r2, s2 = chat(trim(retry)); s += s2; r = r + "\n[retry]\n" + r2; c = c + "\n[retry]\n" + c2; a = parse_action(c2)
            msgs.append({"role": "assistant", "content": c})
            state["turn"] += 1
            worry_scan(c + "\n" + r, "reply")
            with open(f"{LOG}/transcript.md", "a") as f:
                f.write(f"\n\n## turn {state['turn']}  {now():%H:%M:%S}  ({dur(s)} thinking)\n\n### thinking\n{r}\n\n### reply\n{c}\n")
            if not a:
                costs.append((time.time(), "nothing", s))
                log("actions.jsonl", {"turn": state["turn"], "action": None, "think_s": round(s, 1)})
                msgs.append({"role": "user", "content": "(no action was taken)"}); state["last_action_t"] = time.time(); continue
            act = str(a.get("action")); args = a.get("args") or {}
            if not isinstance(args, dict):
                args = {"text": str(args)}
            if act == "stop":
                log("actions.jsonl", {"turn": state["turn"], "action": "stop", "args": args})
                log("attempts.jsonl", {"kind": "STOP_CHOSEN", "reason": str(args.get("reason"))})
                break
            if act not in FN:
                log("attempts.jsonl", {"kind": "unknown_tool", "action": act, "args": {k: str(v)[:500] for k, v in args.items()}})
                res = f"There is no tool called {act}."; dt = 0
            else:
                t = time.time()
                try:
                    res = str(FN[act](args))
                except Exception as e:
                    res = f"That did not work ({type(e).__name__}: {e})"
                dt = time.time() - t
            costs.append((time.time(), act, s + dt))
            log("actions.jsonl", {"turn": state["turn"], "action": act, "args": {k: str(v)[:500] for k, v in args.items()},
                                  "think_s": round(s, 1), "tool_s": round(dt, 1)})
            msgs.append({"role": "user", "content": f"RESULT of {act} (took {dur(dt)}):\n{res[:6000]}"})
            state["last_action_t"] = time.time(); save_state()
    finally:
        stop_server(); save_state()
        if not DRY:
            open(f"{BASE}/NIGHT_OVER", "w").write(now().isoformat())


if __name__ == "__main__":
    main()
